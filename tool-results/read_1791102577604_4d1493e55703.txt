    50→    50→// Isliye: sirf WIDTH change ya height BADHNE par recompute — shrink ignore.
    51→    51→function useRowsPerPage(): number {
    52→    52→  const [rows, setRows] = useState(8);
    53→    53→  useEffect(() => {
    54→    54→    let lastW = 0;
    55→    55→    let lastH = 0;
    56→    56→    const compute = () => {
    57→    57→      const h = window.innerHeight;
    58→    58→      const w = window.innerWidth;
    59→    59→      const widthChanged = lastW !== 0 && Math.abs(w - lastW) > 2;
    60→    60→      const grew = h > lastH;
    61→    61→      if (lastH !== 0 && !widthChanged && !grew) return; // keyboard shrink — ignore
    62→    62→      lastW = w;
    63→    63→      lastH = h;
    64→    64→      const rowH = w >= 640 ? 50 : 46;
    65→    65→      const overhead = w >= 640 ? 385 : 360;
    66→    66→      const n = Math.floor((h - overhead) / rowH);
    67→    67→      setRows(Math.min(18, Math.max(6, n)));
    68→    68→    };
    69→    69→    compute();
    70→    70→    window.addEventListener('resize', compute);
    71→    71→    window.addEventListener('orientationchange', compute);
    72→    72→    return () => {
    73→    73→      window.removeEventListener('resize', compute);
    74→    74→      window.removeEventListener('orientationchange', compute);
    75→    75→    };
    76→    76→  }, []);
    77→    77→  return rows;
    78→    78→}

    93→    93→function HeaderField({
    94→    94→  value,
    95→    95→  display,
    96→    96→  onSave,
    97→    97→  type = 'text',
    98→    98→  ariaLabel,
    99→    99→}: {
   100→   100→  value: string;
   101→   101→  /** what the paper shows (e.g. formatted date) — defaults to value */
   102→   102→  display?: string;
   103→   103→  onSave: (v: string) => void;
   104→   104→  type?: 'text' | 'date';
   105→   105→  ariaLabel: string;
   106→   106→}) {
   107→   107→  const t = useT();
   108→   108→  const [editing, setEditing] = useState(false);
   109→   109→  const [draft, setDraft] = useState(value);
   110→   110→
   111→   111→  const start = () => {
   112→   112→    setDraft(value);
   113→   113→    setEditing(true);
   114→   114→  };
   115→   115→  const commit = () => {
   116→   116→    setEditing(false);
   117→   117→    const v = draft.trim();
   118→   118→    if (v !== value) onSave(v);
   119→   119→  };
   120→   120→
   121→   121→  if (editing) {
   122→   122→    return (
   123→   123→      <input
   124→   124→        autoFocus
   125→   125→        type={type}
   126→   126→        value={draft}
   127→   127→        aria-label={ariaLabel}
   128→   128→        className="cell-input !w-auto max-w-full rounded font-hand text-base font-bold"
   129→   129→        style={{ minWidth: type === 'date' ? '9.5rem' : '4ch' }}
   130→   130→        onChange={(e) => setDraft(e.target.value)}
   131→   131→        onBlur={commit}
   132→   132→        onKeyDown={(e) => {
   133→   133→          if (e.key === 'Enter') {
   134→   134→            e.preventDefault();
   135→   135→            commit();
   136→   136→          }
   137→   137→          if (e.key === 'Escape') {
   138→   138→            e.preventDefault();
   139→   139→            setEditing(false);
   140→   140→          }
   141→   141→        }}
   142→   142→      />
   143→   143→    );
   144→   144→  }
   145→   145→  return (
   146→   146→    <span
   147→   147→      className="paper-blank cursor-text font-hand font-bold decoration-ink/25 underline-offset-4 hover:underline"
   148→   148→      role="button"
   149→   149→      tabIndex={0}
   150→   150→      title={t('headerEditHint')}
   151→   151→      aria-label={`${ariaLabel}: ${display ?? value}`}
   152→   152→      onClick={start}
   153→   153→      onKeyDown={(e) => {
   154→   154→        if (e.key === 'Enter' || e.key === ' ') {
   155→   155→          e.preventDefault();
   156→   156→          start();
   157→   157→        }
   158→   158→      }}
   159→   159→    >
   160→   160→      {display ?? (value || '\u00A0')}
   161→   161→    </span>
   162→   162→  );
   163→   163→}
