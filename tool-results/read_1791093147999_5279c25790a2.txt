    28→    28→function useSuggest() {
    29→    29→  const [sug, setSug] = useState<Suggestion[]>([]);
    30→    30→  const [sugField, setSugField] = useState<SugField>('name');
    31→    31→  const [pick, setPick] = useState<PickPayload | null>(null);
    32→    32→  const seq = useRef(0);
    33→    33→  const wire = (field: SugField) => ({
    34→    34→    onSuggest: (s: Suggestion[]) => {
    35→    35→      setSug(s);
    36→    36→      if (s.length) setSugField(field);
    37→    37→    },
    38→    38→    pick: sugField === field ? pick : null,
    39→    39→  });
    40→    40→  const choose = (sgn: Suggestion) => {
    41→    41→    seq.current += 1;
    42→    42→    setPick({ roman: sgn.roman, text: sgn.text, seq: seq.current });
    43→    43→    setSug([]);
    44→    44→  };
    45→    45→  return { sug, wire, choose, clear: () => setSug([]) };
    46→    46→}

   163→   163→export function WritingRow({
   164→   164→  serial,
   165→   165→  script,
   166→   166→  focusSignal,
   167→   167→  onCommitted,
   168→   168→}: {
   169→   169→  serial: number;
   170→   170→  script: InputScriptSetting;
   171→   171→  focusSignal: number;
   172→   172→  onCommitted: () => void;
   173→   173→}) {
   174→   174→  const t = useT();
   175→   175→  const allEntries = useDaali((s) => s.allEntries);
   176→   176→  const currentEventId = useDaali((s) => s.currentEventId);
   177→   177→  const addEntry = useDaali((s) => s.addEntry);
   178→   178→  const { sug, wire, choose, clear } = useSuggest();
   179→   179→
   180→   180→  const [active, setActive] = useState(false);
   181→   181→  const [name, setName] = useState('');
   182→   182→  const [nameRaw, setNameRaw] = useState('');
   183→   183→  const [village, setVillage] = useState('');
   184→   184→  const [villageRaw, setVillageRaw] = useState('');
   185→   185→  const [relationship, setRelationship] = useState('');
   186→   186→  const [amountRaw, setAmountRaw] = useState('');
   187→   187→  // नेवता/सामान mode — non-cash gift written in place of the amount
   188→   188→  const [itemMode, setItemMode] = useState(false);
   189→   189→  const [itemText, setItemText] = useState('');
   190→   190→  const [date, setDate] = useState(todayISO());
   191→   191→  const [nameErr, setNameErr] = useState('');
   192→   192→  const [amountErr, setAmountErr] = useState('');
   193→   193→  const [dup, setDup] = useState<string | null>(null);
   194→   194→  const [busy, setBusy] = useState(false);
   195→   195→
   196→   196→  const nameRef = useRef<HTMLInputElement>(null);
   197→   197→  const villageMobileRef = useRef<HTMLInputElement>(null);
   198→   198→  const villageDesktopRef = useRef<HTMLInputElement>(null);
   199→   199→  const relationMobileRef = useRef<HTMLInputElement>(null);
   200→   200→  const relationDesktopRef = useRef<HTMLInputElement>(null);
   201→   201→  const amountRef = useRef<HTMLInputElement>(null);
   202→   202→  const itemRef = useRef<HTMLInputElement>(null);
   203→   203→
   204→   204→  // focus the input that is actually visible (mobile vs desktop grids)
   205→   205→  const focusFirstVisible = (...els: Array<HTMLInputElement | null>) => {
   206→   206→    const el = els.find((x) => x && x.offsetParent !== null);
   207→   207→    el?.focus();
   208→   208→  };
   209→   209→  const focusVillage = () => focusFirstVisible(villageMobileRef.current, villageDesktopRef.current);
   210→   210→  const focusRelation = () => focusFirstVisible(relationMobileRef.current, relationDesktopRef.current);
   211→   211→
   212→   212→  // ＋ button / page-advance asks this line to start writing.
   213→   213→  // Only reacts to signal CHANGES (not the value at mount) so that flipping
   214→   214→  // pages never pops the keyboard by itself.
   215→   215→  const initialSignal = useRef(focusSignal);
   216→   216→  useEffect(() => {
   217→   217→    if (focusSignal === initialSignal.current) return;
   218→   218→    initialSignal.current = focusSignal;
   219→   219→    setActive(true);
   220→   220→    const tm = setTimeout(() => nameRef.current?.focus(), 60);
   221→   221→    return () => clearTimeout(tm);
   222→   222→  }, [focusSignal]);
   223→   223→
   224→   224→  const amountNum = parseInt(amountRaw || '0', 10) || 0;
   225→   225→
   226→   226→  const findDuplicate = () =>
   227→   227→    allEntries.find(
   228→   228→      (e) =>
   229→   229→        e.eventId === currentEventId &&
   230→   230→        e.name.trim() !== '' &&
   231→   231→        (normalizeName(e.name) === normalizeName(name) ||
   232→   232→          (!!nameRaw && !!e.nameLatin && normalizeName(e.nameLatin) === normalizeName(nameRaw)))
   233→   233→    );
   234→   234→
   235→   235→  const reset = () => {
   236→   236→    setName('');
   237→   237→    setNameRaw('');
   238→   238→    setVillage('');
   239→   239→    setVillageRaw('');
   240→   240→    setRelationship('');
   241→   241→    setAmountRaw('');
   242→   242→    setItemMode(false);
   243→   243→    setItemText('');
   244→   244→    setDate(todayISO());
   245→   245→    setNameErr('');
   246→   246→    setAmountErr('');
   247→   247→    setDup(null);
   248→   248→    clear();
   249→   249→  };
   250→   250→
   251→   251→  const doCommit = async (force = false) => {
   252→   252→    if (busy) return;
   253→   253→    const nm = name.trim();
   254→   254→    const it = itemMode ? itemText.trim() : '';
   255→   255→    let ok = true;
   256→   256→    if (!nm) {
   257→   257→      setNameErr(t('nameRequired'));
   258→   258→      ok = false;
   259→   259→    } else setNameErr('');
   260→   260→    if (amountNum <= 0 && !it) {
   261→   261→      setAmountErr(t('amountRequired'));
   262→   262→      ok = false;
   263→   263→    } else setAmountErr('');
   264→   264→    if (!ok) {
   265→   265→      (nm ? (itemMode ? itemRef : amountRef) : nameRef).current?.focus();
   266→   266→      return;
   267→   267→    }
   268→   268→    // duplicate — never blocks, just asks once (paper-note style)
   269→   269→    if (!force && findDuplicate()) {
   270→   270→      setDup(nm);
   271→   271→      return;
   272→   272→    }
   273→   273→    setDup(null);
   274→   274→    setBusy(true);
   275→   275→    try {
   276→   276→      await addEntry({
   277→   277→        name: nm,
   278→   278→        village,
   279→   279→        relationship,
   280→   280→        amount: it ? 0 : amountNum,
   281→   281→        item: it,
   282→   282→        date,
   283→   283→        note: '',
   284→   284→        nameLatin: nameRaw !== nm ? nameRaw : '',
   285→   285→        villageLatin: villageRaw !== village ? villageRaw : '',
   286→   286→      });
   287→   287→      // फटाफट — the next line is ready immediately
   288→   288→      reset();
   289→   289→      onCommitted();
   290→   290→      setTimeout(() => nameRef.current?.focus(), 60);
   291→   291→    } catch {
   292→   292→      toast.error(t('saveError'));
   293→   293→    } finally {
   294→   294→      setBusy(false);
   295→   295→    }
   296→   296→  };
   297→   297→
   298→   298→  const cancelLine = () => {
   299→   299→    reset();
   300→   300→    setActive(false);
   301→   301→  };
   302→   302→
   303→   303→  // ── inactive: a quiet empty ruled line with a faint ✎ ──
   304→   304→  if (!active) {
   305→   305→    return (
   306→   306→      <div
   307→   307→        className={`ruled-row ${GRID} cursor-text items-center hover:bg-ink/[0.035]`}
   308→   308→        onClick={() => {
   309→   309→          setActive(true);
   310→   310→          setTimeout(() => nameRef.current?.focus(), 40);
   311→   311→        }}
   312→   312→        role="button"
   313→   313→        tabIndex={0}
   314→   314→        aria-label={t('writeHere')}
   315→   315→        onKeyDown={(e) => {
   316→   316→          if (e.key === 'Enter' || e.key === ' ') {
   317→   317→            e.preventDefault();
   318→   318→            setActive(true);
   319→   319→            setTimeout(() => nameRef.current?.focus(), 40);
   320→   320→          }
   321→   321→        }}
   322→   322→      >
   323→   323→        <div className="text-center text-[13px] text-ink-soft/70">{String(serial).padStart(2, '0')}</div>
   324→   324→        <div className="min-w-0 select-none px-0.5 text-[15px] text-ink-soft/40">✎ {t('writeHere')}</div>
   325→   325→        <div className="hidden sm:block" />
   326→   326→        <div className="hidden sm:block" />
   327→   327→        <div />
   328→   328→        <div className="hidden sm:block" />
   329→   329→      </div>
   330→   330→    );
   331→   331→  }
   332→   332→
   333→   333→  // ── active: write ON the line ──
   334→   334→  return (
   335→   335→    <div className={`ruled-row ${GRID} items-center bg-ink/[0.03]`}>
   336→   336→      <div className="text-center text-[13px] text-ink-soft">{String(serial).padStart(2, '0')}</div>
   337→   337→
   338→   338→      {/* नाम (+ mobile mini-line for गाँव / रिश्ता) */}
   339→   339→      <div className="min-w-0 px-0.5 py-1">
   340→   340→        <div className="flex items-center gap-1">
   341→   341→          <TranslitInput
   342→   342→            ref={nameRef}
   343→   343→            {...wire('name')}
   344→   344→            className="cell-input text-[15px] font-semibold"
   345→   345→            placeholder={t('namePh')}
   346→   346→            value={name}
   347→   347→            maxLength={60}
   348→   348→            autoComplete="off"
   349→   349→            enterKeyHint="next"
   350→   350→            aria-label={t('name')}
   351→   351→            script={script}
   352→   352→            onValueChange={(v) => {

   418→   418→      <div className="hidden min-w-0 px-0.5 sm:block">
