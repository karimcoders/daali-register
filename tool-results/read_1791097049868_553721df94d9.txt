    43→    43→// Dynamic rows-per-page: pages feel like a real register on every screen size
    44→    44→function useRowsPerPage(): number {
    45→    45→  const [rows, setRows] = useState(8);
    46→    46→  useEffect(() => {
    47→    47→    const compute = () => {
    48→    48→      const h = window.innerHeight;
    49→    49→      const w = window.innerWidth;
    50→    50→      const rowH = w >= 640 ? 50 : 46;
    51→    51→      const overhead = w >= 640 ? 385 : 360;
    52→    52→      const n = Math.floor((h - overhead) / rowH);
    53→    53→      setRows(Math.min(18, Math.max(6, n)));
    54→    54→    };
    55→    55→    compute();
    56→    56→    window.addEventListener('resize', compute);
    57→    57→    return () => window.removeEventListener('resize', compute);
    58→    58→  }, []);
    59→    59→  return rows;
    60→    60→}

    74→    74→function HeaderField({
    75→    75→  value,
    76→    76→  display,
    77→    77→  onSave,
    78→    78→  type = 'text',
    79→    79→  ariaLabel,
    80→    80→}: {
    81→    81→  value: string;
    82→    82→  /** what the paper shows (e.g. formatted date) — defaults to value */
    83→    83→  display?: string;
    84→    84→  onSave: (v: string) => void;
    85→    85→  type?: 'text' | 'date';
    86→    86→  ariaLabel: string;
    87→    87→}) {
    88→    88→  const t = useT();
    89→    89→  const [editing, setEditing] = useState(false);
    90→    90→  const [draft, setDraft] = useState(value);
    91→    91→
    92→    92→  const start = () => {
    93→    93→    setDraft(value);
    94→    94→    setEditing(true);
    95→    95→  };
    96→    96→  const commit = () => {
    97→    97→    setEditing(false);
    98→    98→    const v = draft.trim();
    99→    99→    if (v !== value) onSave(v);
   100→   100→  };
   101→   101→
   102→   102→  if (editing) {
   103→   103→    return (
   104→   104→      <input
   105→   105→        autoFocus
   106→   106→        type={type}
   107→   107→        value={draft}
   108→   108→        aria-label={ariaLabel}
   109→   109→        className="cell-input !w-auto max-w-full rounded font-hand text-base font-bold"
   110→   110→        style={{ minWidth: type === 'date' ? '9.5rem' : '4ch' }}
   111→   111→        onChange={(e) => setDraft(e.target.value)}
   112→   112→        onBlur={commit}
   113→   113→        onKeyDown={(e) => {
   114→   114→          if (e.key === 'Enter') {
   115→   115→            e.preventDefault();
   116→   116→            commit();
   117→   117→          }
   118→   118→          if (e.key === 'Escape') {
   119→   119→            e.preventDefault();
   120→   120→            setEditing(false);
   121→   121→          }
   122→   122→        }}
   123→   123→      />
   124→   124→    );
   125→   125→  }
   126→   126→  return (
   127→   127→    <span
   128→   128→      className="paper-blank cursor-text font-hand font-bold decoration-ink/25 underline-offset-4 hover:underline"
   129→   129→      role="button"
   130→   130→      tabIndex={0}
   131→   131→      title={t('headerEditHint')}
   132→   132→      aria-label={`${ariaLabel}: ${display ?? value}`}
   133→   133→      onClick={start}
   134→   134→      onKeyDown={(e) => {
   135→   135→        if (e.key === 'Enter' || e.key === ' ') {
   136→   136→          e.preventDefault();
   137→   137→          start();
   138→   138→        }
   139→   139→      }}
   140→   140→    >
   141→   141→      {display ?? (value || '\u00A0')}
   142→   142→    </span>
   143→   143→  );
   144→   144→}

   147→   147→  const t = useT();

   150→   150→  const currentEventId = useDaali((s) => s.currentEventId);

   154→   154→  const highlightId = useDaali((s) => s.highlightId);

   156→   156→  const setInputScript = useDaali((s) => s.setInputScript);
   157→   157→  const updateEvent = useDaali((s) => s.updateEvent);
   158→   158→
   159→   159→  const rowsPerPage = useRowsPerPage();
   160→   160→  const [flip, setFlip] = useState<'next' | 'prev' | null>(null);
   161→   161→  const [newIds, setNewIds] = useState<Set<string>>(new Set());

   166→   166→  const [writeSignal, setWriteSignal] = useState(0);
   167→   167→  // khali filler line par click → likhai ki line WAHIN chali jati hai (line 5, 6… kahin bhi)

   173→   173→      return !window.localStorage.getItem('daali-edit-hint-v1');

   175→   175→      return false;
   176→   176→    }
   177→   177→  });
   178→   178→  const dismissHint = () => {

   180→   180→    try {
   181→   181→      window.localStorage.setItem('daali-edit-hint-v1', '1');

   184→   184→    }
   185→   185→  };

   192→   192→  );
   193→   193→
   194→   194→  const total = sorted.length;
   195→   195→  // +1: the next empty line (writing slot) counts as a page content line —
   196→   196→  // jaise asli register mein agli khali line hamesha hoti hai

   200→   200→  const pageEntries = sorted.slice(startIdx, startIdx + rowsPerPage);

   202→   202→  const writingHere = writingPage === page;
   203→   203→  const fillerCount = rowsPerPage - pageEntries.length - (writingHere ? 1 : 0);
   204→   204→
   205→   205→  const pageSum = useMemo(() => pageEntries.reduce((a, e) => a + (e.amount || 0), 0), [pageEntries]);
   206→   206→  const totalSum = useMemo(() => sorted.reduce((a, e) => a + (e.amount || 0), 0), [sorted]);
   207→   207→  const itemCount = useMemo(() => countItemEntries(sorted), [sorted]);
   208→   208→  const cashCount = useMemo(() => sorted.filter((e) => e.amount > 0).length, [sorted]);
   209→   209→  const avg = cashCount > 0 ? Math.round(totalSum / cashCount) : 0;

   214→   214→      knownIds.current = new Set(allEntries.map((e) => e.id));
   215→   215→      firstRender.current = false;
   216→   216→      return;
   217→   217→    }
   218→   218→    const fresh = allEntries.filter((e) => !knownIds.current.has(e.id));
   219→   219→    if (fresh.length) {
   220→   220→      for (const f of fresh) knownIds.current.add(f.id);
   221→   221→      setNewIds(new Set(fresh.map((f) => f.id)));
   222→   222→      const timer = setTimeout(() => setNewIds(new Set()), 500);
   223→   223→      return () => clearTimeout(timer);
   224→   224→    }
   225→   225→  }, [allEntries]);
   226→   226→
   227→   227→  const turn = useCallback(
   228→   228→    (dir: 'next' | 'prev') => {
   229→   229→      const target = dir === 'next' ? page + 1 : page - 1;
   230→   230→      if (target < 1 || target > pageCount) return;
   231→   231→      setEditing(null);
   232→   232→      setPage(target);
   233→   233→      if (settings.pageAnimation) {
   234→   234→        setFlip(dir);
   235→   235→        setTimeout(() => setFlip(null), 320);
   236→   236→      }
   237→   237→    },
   238→   238→    [page, pageCount, setPage, settings.pageAnimation]
   239→   239→  );
   240→   240→
   241→   241→  const jumpTo = useCallback(
   242→   242→    (n: number) => {
   243→   243→      const target = Math.min(Math.max(1, n), pageCount);
   244→   244→      if (target !== page && settings.pageAnimation) {
   245→   245→        setFlip(target > page ? 'next' : 'prev');
   246→   246→        setTimeout(() => setFlip(null), 320);
   247→   247→      }
   248→   248→      setEditing(null);
   249→   249→      setPage(target);
   250→   250→    },
   251→   251→    [page, pageCount, setPage, settings.pageAnimation]
   252→   252→  );
   253→   253→
   254→   254→  // kahin bhi khali line par click → wahin pen rakh do (writing line par le jao)
   255→   255→  const focusWriting = useCallback(() => {
   256→   256→    setEditing(null);
   257→   257→    setWriteAtFiller(null);
   258→   258→    if (writingPage !== page) {
   259→   259→      setPage(writingPage);
   260→   260→      setTimeout(() => setWriteSignal((v) => v + 1), settings.pageAnimation ? 360 : 40);
   261→   261→    } else {
   262→   262→      setWriteSignal((v) => v + 1);
   263→   263→    }
   264→   264→  }, [writingPage, page, setPage, settings.pageAnimation]);
   265→   265→
   266→   266→  // ＋ button (bottom nav / top bar) → jump to the writing line and start writing.
   267→   267→  // Arrives as a window event so state updates happen in a callback, not in an effect body.
   268→   268→  useEffect(() => {
   269→   269→    const onWriteFocus = () => focusWriting();
   270→   270→    window.addEventListener('daali:write-focus', onWriteFocus);
   271→   271→    return () => window.removeEventListener('daali:write-focus', onWriteFocus);
   272→   272→  }, [focusWriting]);
   273→   273→
   274→   274→  // after a line is written: if the page became full, palat to the next page automatically
   275→   275→  const onCommitted = useCallback(() => {
   276→   276→    setWriteAtFiller(null); // likh gayi → pen wapas pehli khali line par
   277→   277→    if (pageEntries.length + 1 >= rowsPerPage) {
   278→   278→      setPage(page + 1);
   279→   279→      setTimeout(() => setWriteSignal((v) => v + 1), settings.pageAnimation ? 360 : 40);
   280→   280→    }
   281→   281→  }, [pageEntries.length, rowsPerPage, page, setPage, settings.pageAnimation]);
   282→   282→
   283→   283→  // Keyboard page turning
   284→   284→  useEffect(() => {
   285→   285→    const onKey = (e: KeyboardEvent) => {
   286→   286→      const el = e.target as HTMLElement | null;
   287→   287→      if (el && (el.tagName === 'INPUT' || el.tagName === 'TEXTAREA' || el.tagName === 'SELECT' || el.isContentEditable)) return;
   288→   288→      if (e.metaKey || e.ctrlKey || e.altKey) return;
   289→   289→      if (e.key === 'ArrowRight') turn('next');
   290→   290→      if (e.key === 'ArrowLeft') turn('prev');
   291→   291→    };
   292→   292→    window.addEventListener('keydown', onKey);
   293→   293→    return () => window.removeEventListener('keydown', onKey);
   294→   294→  }, [turn]);
   295→   295→
   296→   296→  // Swipe (mobile) — like palat-ing a page
   297→   297→  const onTouchStart = (e: React.TouchEvent) => {
   298→   298→    const tch = e.touches[0];
   299→   299→    touch.current = { x: tch.clientX, y: tch.clientY };
   300→   300→  };
   301→   301→  const onTouchEnd = (e: React.TouchEvent) => {
   302→   302→    if (!touch.current) return;
   303→   303→    const tch = e.changedTouches[0];
   304→   304→    const dx = tch.clientX - touch.current.x;
   305→   305→    const dy = tch.clientY - touch.current.y;
   306→   306→    touch.current = null;
   307→   307→    if (Math.abs(dx) > 65 && Math.abs(dx) > Math.abs(dy) * 1.6) {
   308→   308→      turn(dx < 0 ? 'next' : 'prev');
   309→   309→    }
   310→   310→  };
   311→   311→
   312→   312→  const flipClass =
   313→   313→    flip === 'next' ? 'page-turn-next' : flip === 'prev' ? 'page-turn-prev' : '';
   314→   314→
   315→   315→  const scriptOptions: Array<{ v: 'hi' | 'ur' | 'off'; label: string }> = [
   316→   316→    { v: 'hi', label: t('scriptHi') },
   317→   317→    { v: 'ur', label: t('scriptUr') },
   318→   318→    { v: 'off', label: t('scriptEn') },
   319→   319→  ];
   320→   320→
   321→   321→  if (!event) return null;
   322→   322→
   323→   323→  return (
   324→   324→    <div className="flex h-full flex-col">
   325→   325→      {/* ── top bar ── */}

   328→   328→          className="ghost-ink-btn flex h-10 w-10 shrink-0 items-center justify-center !border-ink/25"
   329→   329→          onClick={goHome}
   330→   330→          aria-label={t('goBack')}

   332→   332→          <Undo2 className="h-5 w-5" />

   340→   340→          role="group"
   341→   341→          aria-label={t('scriptLabel')}

   349→   349→                  ? 'bg-ink text-[#f7f2e2]'
   350→   350→                  : 'text-ink hover:bg-ink/[0.07]'
   351→   351→              }`}
   352→   352→              onClick={() => setInputScript(o.v)}
   353→   353→              aria-pressed={settings.inputScript === o.v}
   354→   354→              aria-label={`${t('scriptLabel')}: ${o.label}`}
   355→   355→            >
   356→   356→              {o.label}
   357→   357→            </button>
   358→   358→          ))}
   359→   359→        </div>
   360→   360→        <button
   361→   361→          className="ghost-ink-btn flex h-10 w-10 shrink-0 items-center justify-center !border-ink/25"

   363→   363→          aria-label={t('pdfDownload')}
   364→   364→          title={t('pdfDownload')}

   371→   371→              className="ghost-ink-btn flex h-10 w-10 shrink-0 items-center justify-center !border-ink/25"
   372→   372→              aria-label={t('navMore')}

   379→   379→              <Pencil className="h-4 w-4" /> {t('renameEvent')}

   381→   381→            <DropdownMenuItem onClick={onPdf}>
   382→   382→              <FileDown className="h-4 w-4" /> {t('pdfDownload')}
   383→   383→            </DropdownMenuItem>
   384→   384→            <DropdownMenuItem onClick={onHistory}>
   385→   385→              <History className="h-4 w-4" /> {t('viewHistory')}
   386→   386→            </DropdownMenuItem>
   387→   387→            <DropdownMenuItem onClick={onPrint}>
   388→   388→              <Printer className="h-4 w-4" /> {t('printPdf')}
   389→   389→            </DropdownMenuItem>
   390→   390→            <DropdownMenuSeparator />
   391→   391→            <DropdownMenuItem onClick={onDeleteEvent} className="text-destructive focus:text-destructive">
   392→   392→              <Trash2 className="h-4 w-4" /> {t('delete')}

   394→   394→          </DropdownMenuContent>
   395→   395→        </DropdownMenu>
   396→   396→      </div>
   397→   397→
   398→   398→      {/* ── tap-to-edit hint (first visit only) ── */}
   399→   399→      {showHint && (
   400→   400→        <div className="no-print mx-2 mt-1 flex items-center gap-2 rounded-md border border-dashed border-ink/30 bg-[var(--paper-2)] px-3 py-1.5 sm:mx-6">

   402→   402→          <p className="min-w-0 flex-1 text-[12.5px] leading-snug text-ink-soft">{t('editHint')}</p>

   404→   404→            {t('editHintOk')}
   405→   405→          </button>
   406→   406→        </div>
   407→   407→      )}
   408→   408→
   409→   409→      {/* ── scrollable register area ── */}
   410→   410→      <div
   411→   411→        className="no-print flex flex-1 items-start justify-center overflow-y-auto px-2 py-2 sm:px-6 sm:py-3"
   412→   412→        onTouchStart={onTouchStart}
   413→   413→        onTouchEnd={onTouchEnd}
   414→   414→        style={{ scrollbarWidth: 'thin' }}
   415→   415→      >
   416→   416→        <div
   417→   417→          key={page}
   418→   418→          className={`paper flex min-h-full w-full max-w-3xl flex-col pb-2 ${flipClass}`}
   419→   419→          role="region"
   420→   420→          aria-label={`${t('daaliRegister')} — ${event.name}`}
   421→   421→        >
   422→   422→          <div className="paper-spine" />
   423→   423→          <div className="paper-margin" />
   424→   424→
   425→   425→          {/* page number */}
   426→   426→          <div className="absolute end-3 top-2 font-hand text-xs text-ink-soft">
   427→   427→            {t('page')} {page}
   428→   428→          </div>

   430→   430→          {/* header — like a real register heading */}
   431→   431→          <header className="px-4 pt-3 text-center sm:px-8">

   613→   613→function RegisterRow({
