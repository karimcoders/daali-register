    28→function useSuggest() {
    29→  const [sug, setSug] = useState<Suggestion[]>([]);
    30→  const [sugField, setSugField] = useState<SugField>('name');
    31→  const [pick, setPick] = useState<PickPayload | null>(null);
    32→  const seq = useRef(0);
    33→  const wire = (field: SugField) => ({
    34→    onSuggest: (s: Suggestion[]) => {
    35→      setSug(s);
    36→      if (s.length) setSugField(field);
    37→    },
    38→    pick: sugField === field ? pick : null,
    39→  });
    40→  const choose = (sgn: Suggestion) => {
    41→    seq.current += 1;
    42→    setPick({ roman: sgn.roman, text: sgn.text, seq: seq.current });
    43→    setSug([]);
    44→  };
    45→  return { sug, wire, choose, clear: () => setSug([]) };
    46→}

    48→// dictionary suggestion chips — tap to write the word perfectly
    49→function SuggestionChips({
    50→  sug,
    51→  onChoose,
    52→}: {
    53→  sug: Suggestion[];
    54→  onChoose: (s: Suggestion) => void;
    55→}) {
    56→  if (!sug.length) return null;
    57→  return (
    58→    <div className="col-span-full flex flex-wrap items-center gap-1 pb-1 pt-0.5">
    59→      <span className="mr-0.5 text-[11px] text-ink-soft/60">✎</span>
    60→      {sug.map((sgn) => (
    61→        <button
    62→          key={sgn.roman}
    63→          type="button"
    64→          className="chip h-8 px-2.5 text-[14px]"
    65→          onMouseDown={(e) => e.preventDefault()}
    66→          onTouchStart={(e) => e.preventDefault()}
    67→          onClick={() => onChoose(sgn)}
    68→        >
    69→          {sgn.text}
    70→        </button>
    71→      ))}
    72→    </div>
    73→  );
    74→}

    76→// quick ₹ chips — shown only while the line is active
    77→function QuickAmounts({
    78→  amountRaw,
    79→  itemText,
    80→  itemMode,
    81→  onMode,
    82→  onPick,
    83→}: {
    84→  amountRaw: string;
    85→  itemText: string;
    86→  itemMode: boolean;
    87→  onMode: (v: boolean) => void;
    88→  onPick: (n: number) => void;
    89→}) {
    90→  const t = useT();
    91→  const amountNum = parseInt(amountRaw || '0', 10) || 0;
    92→  return (
    93→    <div className="col-span-full flex flex-wrap items-center gap-1 pb-1 pt-0.5">
    94→      <ModeChips
    95→        itemMode={itemMode}
    96→        onMode={onMode}
    97→        label={t('amount')}
    98→      />
    99→      {!itemMode &&
   100→        QUICK_AMOUNTS.map((q) => (
   101→          <button
   102→            key={q}
   103→            type="button"
   104→            className={`chip h-8 min-w-[3.1rem] px-2 text-[13px] ${String(q) === amountRaw ? '!border-margin-red !bg-margin-red/10 font-bold' : ''}`}
   105→            onClick={() => onPick(q)}
   106→            aria-label={formatRupees(q)}
   107→          >
   108→            {formatRupees(q)}
   109→          </button>
   110→        ))}
   111→      {itemMode ? (
   112→        itemText.trim() ? (
   113→          <span className="ml-auto font-hand text-lg font-bold text-ink">🎁 {itemText.trim()}</span>
   114→        ) : null
   115→      ) : (
   116→        amountNum > 0 && (
   117→          <span className="ml-auto font-hand text-lg font-bold text-ink">{formatRupees(amountNum)}</span>
   118→        )
   119→      )}
   120→    </div>
   121→  );
   122→}

   162→// ─── the next empty line — writing happens here ─────────────────────────────
   163→export function WritingRow({
   164→  serial,
   165→  script,
   166→  focusSignal,
   167→  onCommitted,
   168→}: {
   169→  serial: number;
   170→  script: InputScriptSetting;
   171→  focusSignal: number;
   172→  onCommitted: () => void;
   173→}) {
   174→  const t = useT();
   175→  const allEntries = useDaali((s) => s.allEntries);
   176→  const currentEventId = useDaali((s) => s.currentEventId);
   177→  const addEntry = useDaali((s) => s.addEntry);
   178→  const { sug, wire, choose, clear } = useSuggest();
   179→
   180→  const [active, setActive] = useState(false);
   181→  const [name, setName] = useState('');
   182→  const [nameRaw, setNameRaw] = useState('');
   183→  const [village, setVillage] = useState('');
   184→  const [villageRaw, setVillageRaw] = useState('');
   185→  const [relationship, setRelationship] = useState('');
   186→  const [amountRaw, setAmountRaw] = useState('');
   187→  // नेवता/सामान mode — non-cash gift written in place of the amount
   188→  const [itemMode, setItemMode] = useState(false);
   189→  const [itemText, setItemText] = useState('');
   190→  const [date, setDate] = useState(todayISO());
   191→  const [nameErr, setNameErr] = useState('');
   192→  const [amountErr, setAmountErr] = useState('');
   193→  const [dup, setDup] = useState<string | null>(null);
   194→  const [busy, setBusy] = useState(false);
   195→
   196→  const nameRef = useRef<HTMLInputElement>(null);
   197→  const villageMobileRef = useRef<HTMLInputElement>(null);
   198→  const villageDesktopRef = useRef<HTMLInputElement>(null);
   199→  const relationMobileRef = useRef<HTMLInputElement>(null);
   200→  const relationDesktopRef = useRef<HTMLInputElement>(null);
   201→  const amountRef = useRef<HTMLInputElement>(null);
   202→  const itemRef = useRef<HTMLInputElement>(null);
   203→
   204→  // focus the input that is actually visible (mobile vs desktop grids)
   205→  const focusFirstVisible = (...els: Array<HTMLInputElement | null>) => {
   206→    const el = els.find((x) => x && x.offsetParent !== null);
   207→    el?.focus();
   208→  };
   209→  const focusVillage = () => focusFirstVisible(villageMobileRef.current, villageDesktopRef.current);
   210→  const focusRelation = () => focusFirstVisible(relationMobileRef.current, relationDesktopRef.current);
   211→
   212→  // ＋ button / page-advance asks this line to start writing.
   213→  // Only reacts to signal CHANGES (not the value at mount) so that flipping
   214→  // pages never pops the keyboard by itself.
   215→  const initialSignal = useRef(focusSignal);
   216→  useEffect(() => {
   217→    if (focusSignal === initialSignal.current) return;
   218→    initialSignal.current = focusSignal;
   219→    setActive(true);
   220→    const tm = setTimeout(() => nameRef.current?.focus(), 60);
   221→    return () => clearTimeout(tm);
   222→  }, [focusSignal]);
   223→
   224→  const amountNum = parseInt(amountRaw || '0', 10) || 0;
   225→
   226→  const findDuplicate = () =>
   227→    allEntries.find(
   228→      (e) =>
   229→        e.eventId === currentEventId &&
   230→        e.name.trim() !== '' &&
   231→        (normalizeName(e.name) === normalizeName(name) ||
   232→          (!!nameRaw && !!e.nameLatin && normalizeName(e.nameLatin) === normalizeName(nameRaw)))
   233→    );
   234→
   235→  const reset = () => {
   236→    setName('');
   237→    setNameRaw('');
   238→    setVillage('');
   239→    setVillageRaw('');
   240→    setRelationship('');
   241→    setAmountRaw('');
   242→    setItemMode(false);
   243→    setItemText('');
   244→    setDate(todayISO());
   245→    setNameErr('');
   246→    setAmountErr('');
   247→    setDup(null);
   248→    clear();
   249→  };
   250→
   251→  const doCommit = async (force = false) => {
   252→    if (busy) return;
   253→    const nm = name.trim();
   254→    const it = itemMode ? itemText.trim() : '';
   255→    let ok = true;
   256→    if (!nm) {
   257→      setNameErr(t('nameRequired'));
   258→      ok = false;
   259→    } else setNameErr('');
   260→    if (amountNum <= 0 && !it) {
   261→      setAmountErr(t('amountRequired'));
   262→      ok = false;
   263→    } else setAmountErr('');
   264→    if (!ok) {
   265→      (nm ? (itemMode ? itemRef : amountRef) : nameRef).current?.focus();
   266→      return;
   267→    }
   268→    // duplicate — never blocks, just asks once (paper-note style)
   269→    if (!force && findDuplicate()) {
   270→      setDup(nm);
   271→      return;
   272→    }
   273→    setDup(null);
   274→    setBusy(true);
   275→    try {
   276→      await addEntry({
   277→        name: nm,
   278→        village,
   279→        relationship,
   280→        amount: it ? 0 : amountNum,
   281→        item: it,
   282→        date,
   283→        note: '',
   284→        nameLatin: nameRaw !== nm ? nameRaw : '',
   285→        villageLatin: villageRaw !== village ? villageRaw : '',
   286→      });
   287→      // फटाफट — the next line is ready immediately
   288→      reset();
   289→      onCommitted();
   290→      setTimeout(() => nameRef.current?.focus(), 60);
   291→    } catch {
   292→      toast.error(t('saveError'));
   293→    } finally {
   294→      setBusy(false);
   295→    }
   296→  };
   297→
   298→  const cancelLine = () => {
   299→    reset();
   300→    setActive(false);
   301→  };
   302→
   303→  // ── inactive: a quiet empty ruled line with a faint ✎ ──
   304→  if (!active) {
   305→    return (
   306→      <div
   307→        className={`ruled-row ${GRID} cursor-text items-center hover:bg-ink/[0.035]`}
   308→        onClick={() => {
   309→          setActive(true);
   310→          setTimeout(() => nameRef.current?.focus(), 40);
   311→        }}
   312→        role="button"
   313→        tabIndex={0}
   314→        aria-label={t('writeHere')}
   315→        onKeyDown={(e) => {
   316→          if (e.key === 'Enter' || e.key === ' ') {
   317→            e.preventDefault();
   318→            setActive(true);
   319→            setTimeout(() => nameRef.current?.focus(), 40);
   320→          }
   321→        }}
   322→      >
   323→        <div className="text-center text-[13px] text-ink-soft/70">{String(serial).padStart(2, '0')}</div>
   324→        <div className="min-w-0 select-none px-0.5 text-[15px] text-ink-soft/40">✎ {t('writeHere')}</div>
   325→        <div className="hidden sm:block" />
   326→        <div className="hidden sm:block" />
   327→        <div />
   328→        <div className="hidden sm:block" />
   329→      </div>
   330→    );
   331→  }
   332→
   333→  // ── active: write ON the line ──
   334→  return (
   335→    <div className={`ruled-row ${GRID} items-center bg-ink/[0.03]`}>
   336→      <div className="text-center text-[13px] text-ink-soft">{String(serial).padStart(2, '0')}</div>
   337→
   338→      {/* नाम (+ mobile mini-line for गाँव / रिश्ता) */}
   339→      <div className="min-w-0 px-0.5 py-1">
   340→        <div className="flex items-center gap-1">
   341→          <TranslitInput
   342→            ref={nameRef}
   343→            {...wire('name')}
   344→            className="cell-input text-[15px] font-semibold"
   345→            placeholder={t('namePh')}
   346→            value={name}
   347→            maxLength={60}
   348→            autoComplete="off"
   349→            enterKeyHint="next"
   350→            aria-label={t('name')}
   351→            script={script}
   352→            onValueChange={(v) => {
   353→              setName(v);
   354→              setNameErr('');
   355→            }}
   356→            onRawChange={setNameRaw}
   357→            onKeyDown={(e, fv) => {
   358→              if (e.key === 'Enter') {
   359→                e.preventDefault();
   360→                if (fv !== undefined) setName(fv);
   361→                focusVillage();
   362→              }
   363→              if (e.key === 'Escape') {
   364→                e.preventDefault();
   365→                cancelLine();
   366→              }
   367→            }}
   368→          />
   369→          <FieldError msg={nameErr} />
   370→        </div>
   371→        {/* mobile: गाँव + रिश्ता on the second small line */}
   372→        <div className="mt-0.5 flex gap-2 sm:hidden">
   373→          <TranslitInput
   374→            ref={villageMobileRef}
   375→            {...wire('village')}
   376→            className="cell-input min-w-0 flex-1 text-[13px] text-ink-soft"
   377→            placeholder={t('villagePh')}
   378→            value={village}
   379→            maxLength={40}
   380→            autoComplete="off"
   381→            aria-label={t('village')}
   382→            script={script}
   383→            onValueChange={(v) => {
   384→              setVillage(v);
   385→              setVillageRaw(v);
   386→            }}
   387→            onKeyDown={(e) => {
   388→              if (e.key === 'Enter') {
   389→                e.preventDefault();
   390→                focusRelation();
   391→              }
   392→              if (e.key === 'Escape') cancelLine();
   393→            }}
   394→          />
   395→          <TranslitInput
   396→            ref={relationMobileRef}
   397→            {...wire('relation')}
   398→            className="cell-input min-w-0 flex-1 text-[13px] text-ink-soft"
   399→            placeholder={t('relationPh')}
   400→            value={relationship}
   401→            maxLength={40}
   402→            autoComplete="off"
   403→            aria-label={t('relation')}
   404→            script={script}
   405→            onValueChange={setRelationship}
   406→            onKeyDown={(e) => {
   407→              if (e.key === 'Enter') {
   408→                e.preventDefault();
   409→                amountRef.current?.focus();
   410→              }
   411→              if (e.key === 'Escape') cancelLine();
   412→            }}
   413→          />
   414→        </div>
   415→      </div>
   416→
   417→      {/* desktop: गाँव */}
   418→      <div className="hidden min-w-0 px-0.5 sm:block">
   419→        <TranslitInput
   420→          ref={villageDesktopRef}
   421→          {...wire('village')}
   422→          className="cell-input text-sm"
   423→          placeholder={t('villagePh')}
   424→          value={village}
   425→          maxLength={40}
   426→          autoComplete="off"
   427→          aria-label={t('village')}
   428→          script={script}
   429→          onValueChange={(v) => {
   430→            setVillage(v);
   431→            setVillageRaw(v);
   432→          }}
   433→          onKeyDown={(e) => {
   434→            if (e.key === 'Enter') {
   435→              e.preventDefault();
   436→              focusRelation();
   437→            }
   438→            if (e.key === 'Escape') cancelLine();
   439→          }}
   440→        />
   441→      </div>
   442→
   443→      {/* desktop: रिश्ता */}
   444→      <div className="hidden min-w-0 px-0.5 sm:block">
   445→        <TranslitInput
   446→          ref={relationDesktopRef}
   447→          {...wire('relation')}
   448→          className="cell-input text-sm"
   449→          placeholder={t('relationPh')}
   450→          value={relationship}
   451→          maxLength={40}
   452→          autoComplete="off"
   453→          aria-label={t('relation')}
   454→          script={script}
   455→          onValueChange={setRelationship}
   456→          onKeyDown={(e) => {
   457→            if (e.key === 'Enter') {
   458→              e.preventDefault();
   459→              amountRef.current?.focus();
   460→            }
   461→            if (e.key === 'Escape') cancelLine();
   462→          }}
   463→        />
   464→      </div>
   465→
   466→      {/* रकम — ya फिर नेवता/सामान (item mode) */}
   467→      <div className="px-0.5">
   468→        <div className="flex items-center justify-end gap-0.5">
   469→          {itemMode ? (
   470→            <span className="text-base" aria-hidden="true">🎁</span>
   471→          ) : (
   472→            <span className="font-hand text-base font-bold text-ink-soft">₹</span>
   473→          )}
   474→          {itemMode ? (
   475→            <TranslitInput
   476→              ref={itemRef}
   477→              {...wire('item')}
   478→              className="cell-input w-full text-right text-[14px]"
   479→              placeholder={t('itemPh')}
   480→              value={itemText}
   481→              maxLength={60}
   482→              autoComplete="off"
   483→              enterKeyHint="done"
   484→              aria-label={t('itemMode')}
   485→              script={script}
   486→              onValueChange={(v) => {
   487→                setItemText(v);
   488→                setAmountErr('');
   489→              }}
   490→              onKeyDown={(e, fv) => {
   491→                if (e.key === 'Enter') {
   492→                  e.preventDefault();
   493→                  if (fv !== undefined) setItemText(fv);
   494→                  doCommit();
   495→                }
   496→                if (e.key === 'Escape') {
   497→                  e.preventDefault();
   498→                  cancelLine();
   499→                }
   500→              }}
   501→            />
   502→          ) : (
   503→            <input
   504→              ref={amountRef}
   505→              className="cell-input w-full text-right text-[15px] font-bold tabular-nums"
   506→              style={{ fontVariantNumeric: 'tabular-nums' }}
   507→              inputMode="numeric"
   508→              pattern="[0-9]*"
   509→              autoComplete="off"
   510→              placeholder="0"
   511→              value={amountRaw}
   512→              enterKeyHint="done"
   513→              aria-label={t('amount')}
   514→              onChange={(e) => {
   515→                setAmountRaw(sanitizeAmountInput(e.target.value));
   516→                setAmountErr('');
   517→              }}
   518→              onKeyDown={(e) => {
   519→                if (isIMEComposing(e)) return;
   520→                if (e.key === 'Enter') {
   521→                  e.preventDefault();
   522→                  doCommit();
   523→                }
   524→                if (e.key === 'Escape') {
   525→                  e.preventDefault();
   526→                  cancelLine();
   527→                }
   528→              }}
   529→            />
   530→          )}
   531→          <FieldError msg={amountErr} />
   532→        </div>
   533→      </div>
   534→
   535→      {/* desktop: तारीख */}
   536→      <div className="hidden px-0.5 sm:block">
   537→        <input
   538→          type="date"
   539→          className="cell-input text-right text-xs text-ink-soft"
   540→          value={date}
   541→          aria-label={t('date')}
   542→          onChange={(e) => setDate(e.target.value)}
   543→          onKeyDown={(e) => {
   544→            if (isIMEComposing(e)) return;
   545→            if (e.key === 'Enter') {
   546→              e.preventDefault();
   547→              doCommit();
   548→            }
   549→            if (e.key === 'Escape') cancelLine();
   550→          }}
   551→        />
   552→      </div>
   553→
   554→      {/* quick ₹ chips + नेवता toggle + duplicate note — only while writing */}
   555→      {dup ? (
   556→        <div className="col-span-full animate-stamp-in rounded-md border-2 border-dashed border-margin-red/60 bg-margin-red/5 px-3 py-2">
   557→          <p className="text-sm text-ink">
   558→            {t('duplicateWarning')} <span className="font-hand font-bold">{dup}</span>
   559→          </p>
   560→          <div className="mt-1.5 flex gap-2">
   561→            <button
   562→              className="ghost-ink-btn h-9 flex-1 text-base"
   563→              onClick={() => {
   564→                setDup(null);
   565→                nameRef.current?.focus();
   566→              }}
   567→            >
   568→              {t('no')}
   569→            </button>
   570→            <button className="ink-btn h-9 flex-1 text-base" onClick={() => doCommit(true)}>
   571→              {t('writeAnyway')}
   572→            </button>
   573→          </div>
   574→        </div>
   575→      ) : sug.length > 0 ? (
   576→        <SuggestionChips sug={sug} onChoose={choose} />
   577→      ) : (
   578→        <QuickAmounts
   579→          amountRaw={amountRaw}
   580→          itemText={itemText}
   581→          itemMode={itemMode}
   582→          onMode={(v) => {
   583→            setItemMode(v);
   584→            setAmountErr('');
   585→            setTimeout(() => (v ? itemRef : amountRef).current?.focus(), 30);
   586→          }}
   587→          onPick={(q) => {
   588→            setAmountRaw(String(q));
   589→            setAmountErr('');
   590→            nameRef.current?.focus();
   591→          }}
   592→        />
   593→      )}
   594→    </div>
   595→  );
   596→}

   598→// ─── tap a written line → edit it right there ────────────────────────────────
   599→export function EntryEditRow({
   600→  entry,
   601→  serial,
   602→  script,
   603→  focusCell,
   604→  onDone,
   605→}: {
   606→  entry: DaaliEntry;
   607→  serial: number;
   608→  script: InputScriptSetting;
   609→  /** the cell that was tapped — cursor starts there, pen-style */
   610→  focusCell?: EditCell;
   611→  onDone: () => void;
   612→}) {
   613→  const t = useT();
   614→  const allEntries = useDaali((s) => s.allEntries);
   615→  const updateEntry = useDaali((s) => s.updateEntry);
   616→  const removeEntry = useDaali((s) => s.removeEntry);
   617→  const { sug, wire, choose } = useSuggest();
   618→
   619→  const [name, setName] = useState(entry.name);
   620→  const [nameRaw, setNameRaw] = useState(entry.nameLatin || '');
   621→  const [village, setVillage] = useState(entry.village);
   622→  const [villageRaw, setVillageRaw] = useState(entry.villageLatin || '');
   623→  const [relationship, setRelationship] = useState(entry.relationship);
   624→  const [amountRaw, setAmountRaw] = useState(entry.amount ? String(entry.amount) : '');
   625→  const [itemMode, setItemMode] = useState(!(entry.amount > 0));
   626→  const [itemText, setItemText] = useState(entry.item || '');
   627→  const [date, setDate] = useState(entry.date || todayISO());
   628→  const [note, setNote] = useState(entry.note);
   629→  const [nameErr, setNameErr] = useState('');
   630→  const [amountErr, setAmountErr] = useState('');
   631→  const [dup, setDup] = useState<string | null>(null);
   632→  const [confirmDelete, setConfirmDelete] = useState(false);
   633→  const [busy, setBusy] = useState(false);
   634→
   635→  const nameRef = useRef<HTMLInputElement>(null);
   636→  const villageMobileRef = useRef<HTMLInputElement>(null);
   637→  const villageDesktopRef = useRef<HTMLInputElement>(null);
   638→  const relationMobileRef = useRef<HTMLInputElement>(null);
   639→  const relationDesktopRef = useRef<HTMLInputElement>(null);
   640→  const amountRef = useRef<HTMLInputElement>(null);
   641→  const itemRef = useRef<HTMLInputElement>(null);
   642→  const dateRef = useRef<HTMLInputElement>(null);
   643→
   644→  const focusFirstVisible = (...els: Array<HTMLInputElement | null>) => {
   645→    const el = els.find((x) => x && x.offsetParent !== null);
   646→    el?.focus();
   647→  };
   648→  const focusAmount = () => focusFirstVisible(amountRef.current);
   649→
   650→  // cursor starts on the tapped cell — jaise pen wahi rakha ho
   651→  useEffect(() => {
   652→    const tm = setTimeout(() => {
   653→      switch (focusCell) {
   654→        case 'amount':
   655→          amountRef.current?.focus();
   656→          break;
   657→        case 'date':
   658→          dateRef.current?.focus();
   659→          break;
   660→        case 'village':
   661→          focusFirstVisible(villageMobileRef.current, villageDesktopRef.current);
   662→          break;
   663→        case 'relation':
   664→          focusFirstVisible(relationMobileRef.current, relationDesktopRef.current);
   665→          break;
   666→        default:
   667→          nameRef.current?.focus();
   668→      }
   669→    }, 40);
   670→    return () => clearTimeout(tm);
   671→  }, []);
   672→
   673→  const amountNum = parseInt(amountRaw || '0', 10) || 0;
   674→
   675→  const findDuplicate = () =>
   676→    allEntries.find(
   677→      (e) =>
   678→        e.id !== entry.id &&
   679→        e.eventId === entry.eventId &&
   680→        e.name.trim() !== '' &&
   681→        (normalizeName(e.name) === normalizeName(name) ||
   682→          (!!nameRaw && !!e.nameLatin && normalizeName(e.nameLatin) === normalizeName(nameRaw)))
   683→    );
   684→
   685→  const doSave = async (force = false) => {
   686→    if (busy) return;
   687→    const nm = name.trim();
   688→    const it = itemMode ? itemText.trim() : '';
   689→    let ok = true;
   690→    if (!nm) {
   691→      setNameErr(t('nameRequired'));
   692→      ok = false;
   693→    } else setNameErr('');
   694→    if (amountNum <= 0 && !it) {
   695→      setAmountErr(t('amountRequired'));
   696→      ok = false;
   697→    } else setAmountErr('');
   698→    if (!ok) {
   699→      (nm ? (itemMode ? itemRef : amountRef) : nameRef).current?.focus();
   700→      return;
   701→    }
   702→    if (!force && findDuplicate()) {
   703→      setDup(nm);
   704→      return;
   705→    }
   706→    setDup(null);
   707→    setBusy(true);
   708→    try {
   709→      await updateEntry(entry.id, {
   710→        name: nm,
   711→        village,
   712→        relationship,
   713→        amount: it ? 0 : amountNum,
   714→        item: it,
   715→        date,
   716→        note,
   717→        nameLatin: nameRaw !== nm ? nameRaw : '',
   718→        villageLatin: villageRaw !== village ? villageRaw : '',
   719→      });
   720→      toast.success(t('updatedToast'), { duration: 1500 });
   721→      onDone();
   722→    } catch {
   723→      toast.error(t('saveError'));
   724→    } finally {
   725→      setBusy(false);
   726→    }
   727→  };
   728→
   729→  return (
   730→    <div className={`ruled-row ${GRID} items-center bg-ink/[0.04]`}>
   731→      <div className="text-center text-[13px] text-ink-soft">{String(serial).padStart(2, '0')}</div>
   732→
   733→      <div className="min-w-0 px-0.5 py-1">
   734→        <div className="flex items-center gap-1">
   735→          <TranslitInput
   736→            ref={nameRef}
   737→            {...wire('name')}
   738→            className="cell-input text-[15px] font-semibold"
   739→            placeholder={t('namePh')}
   740→            value={name}
   741→            maxLength={60}
   742→            autoComplete="off"
   743→            aria-label={t('name')}
   744→            script={script}
   745→            onValueChange={(v) => {
   746→              setName(v);
   747→              setNameErr('');
   748→            }}
   749→            onRawChange={setNameRaw}
   750→            onKeyDown={(e, fv) => {
   751→              if (e.key === 'Enter') {
   752→                e.preventDefault();
   753→                if (fv !== undefined) setName(fv);
   754→                focusAmount();
   755→              }
   756→              if (e.key === 'Escape') {
   757→                e.preventDefault();
   758→                onDone();
   759→              }
   760→            }}
   761→          />
   762→          <FieldError msg={nameErr} />
   763→        </div>
   764→        {/* mobile: गाँव • रिश्ता second line */}
   765→        <div className="mt-0.5 flex gap-2 sm:hidden">
   766→          <TranslitInput
   767→            ref={villageMobileRef}
   768→            {...wire('village')}
   769→            className="cell-input min-w-0 flex-1 text-[13px] text-ink-soft"
   770→            placeholder={t('villagePh')}
   771→            value={village}
   772→            maxLength={40}
   773→            autoComplete="off"
   774→            aria-label={t('village')}
   775→            script={script}
   776→            onValueChange={(v) => {
   777→              setVillage(v);
   778→              setVillageRaw(v);
   779→            }}
   780→          />
   781→          <TranslitInput
   782→            className="cell-input min-w-0 flex-1 text-[13px] text-ink-soft"
   783→            placeholder={t('relationPh')}
   784→            value={relationship}
   785→            maxLength={40}
   786→            autoComplete="off"
   787→            aria-label={t('relation')}
   788→            script={script}
   789→            onValueChange={setRelationship}
   790→          />
   791→        </div>
   792→      </div>
   793→
   794→      <div className="hidden min-w-0 px-0.5 sm:block">
   795→        <TranslitInput
   796→          className="cell-input text-sm"
   797→          placeholder={t('villagePh')}
   798→          value={village}
   799→          maxLength={40}
   800→          autoComplete="off"
   801→          aria-label={t('village')}
   802→          script={script}
   803→          onValueChange={(v) => {
   804→            setVillage(v);
   805→            setVillageRaw(v);
   806→          }}
   807→        />
   808→      </div>
   809→
   810→      <div className="hidden min-w-0 px-0.5 sm:block">
   811→        <TranslitInput
   812→          className="cell-input text-sm"
   813→          placeholder={t('relationPh')}
   814→          value={relationship}
   815→          maxLength={40}
   816→          autoComplete="off"
   817→          aria-label={t('relation')}
   818→          script={script}
   819→          onValueChange={setRelationship}
   820→        />
   821→      </div>
   822→
   823→      <div className="px-0.5">
   824→        <div className="flex items-center justify-end gap-0.5">
   825→          {itemMode ? (
   826→            <span className="text-base" aria-hidden="true">🎁</span>
   827→          ) : (
   828→            <span className="font-hand text-base font-bold text-ink-soft">₹</span>
   829→          )}
   830→          {itemMode ? (
   831→            <TranslitInput
   832→              ref={itemRef}
   833→              {...wire('item')}
   834→              className="cell-input w-full text-right text-[14px]"
   835→              placeholder={t('itemPh')}
   836→              value={itemText}
   837→              maxLength={60}
   838→              autoComplete="off"
   839→              enterKeyHint="done"
   840→              aria-label={t('itemMode')}
   841→              script={script}
   842→              onValueChange={(v) => {
   843→                setItemText(v);
   844→                setAmountErr('');
   845→              }}
   846→              onKeyDown={(e, fv) => {
   847→                if (e.key === 'Enter') {
   848→                  e.preventDefault();
   849→                  if (fv !== undefined) setItemText(fv);
   850→                  doSave();

   853→                  e.preventDefault();
   854→                  onDone();

   859→            <input
   860→              ref={amountRef}
   861→              className="cell-input w-full text-right text-[15px] font-bold tabular-nums"

   863→              inputMode="numeric"
   864→              pattern="[0-9]*"
   865→              autoComplete="off"
   866→              placeholder="0"
   867→              value={amountRaw}

   871→                setAmountRaw(sanitizeAmountInput(e.target.value));
   872→                setAmountErr('');

   875→                if (isIMEComposing(e)) return;
   876→                if (e.key === 'Enter') {

   891→      <div className="hidden px-0.5 sm:block">

   914→      <div className="col-span-full flex flex-wrap items-center gap-1.5 pb-1.5 pt-0.5">
   915→        <ModeChips
   916→          itemMode={itemMode}
   917→          onMode={(v) => {
   918→            setItemMode(v);
   919→            setAmountErr('');

   921→          label={t('amount')}

   962→              className="ghost-ink-btn h-9 flex-1 text-base"
   963→              onClick={() => {
   964→                setDup(null);
   965→                nameRef.current?.focus();

   968→              {t('no')}
   969→            </button>

   997→          setConfirmDelete(false);
   998→          onDone();
   999→        }}

  1001→    </div>
