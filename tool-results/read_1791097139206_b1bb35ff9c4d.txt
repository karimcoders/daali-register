    28→ * kyunki blur ke waqt agla focus abhi set nahi hua hota.

    30→function focusLeftRow(row: HTMLElement | null): boolean {
    31→  const ae = document.activeElement;
    32→  if (!ae || ae === document.body) return true;
    33→  return !(row && ae instanceof Node && row.contains(ae));
    34→}

    38→function useSuggest() {
    39→  const [sug, setSug] = useState<Suggestion[]>([]);
    40→  const [sugField, setSugField] = useState<SugField>('name');
    41→  const [pick, setPick] = useState<PickPayload | null>(null);
    42→  const seq = useRef(0);
    43→  const wire = (field: SugField) => ({
    44→    onSuggest: (s: Suggestion[]) => {
    45→      setSug(s);
    46→      if (s.length) setSugField(field);
    47→    },
    48→    pick: sugField === field ? pick : null,
    49→  });
    50→  const choose = (sgn: Suggestion) => {
    51→    seq.current += 1;
    52→    setPick({ roman: sgn.roman, text: sgn.text, seq: seq.current });
    53→    setSug([]);
    54→  };
    55→  return { sug, wire, choose, clear: () => setSug([]) };
    56→}

   190→// ─── the next empty line — writing happens here ─────────────────────────────
   191→export function WritingRow({
   192→  serial,
   193→  script,
   194→  focusSignal,
   195→  onCommitted,
   196→  onCancel,
   197→  autoStart,
   198→}: {
   199→  serial: number;
   200→  script: InputScriptSetting;
   201→  focusSignal: number;
   202→  onCommitted: () => void;
   203→  /** ✕/Escape — line chhod do, kuch bhi likha nahi jayega */
   204→  onCancel?: () => void;
   205→  /** khali line par click → pen WAHIN rakha — row turant khuli, likhna shuru */
   206→  autoStart?: boolean;
   207→}) {
   208→  const t = useT();
   209→  const allEntries = useDaali((s) => s.allEntries);
   210→  const currentEventId = useDaali((s) => s.currentEventId);
   211→  const addEntry = useDaali((s) => s.addEntry);
   212→  const { sug, wire, choose, clear } = useSuggest();
   213→
   214→  const [active, setActive] = useState(autoStart ?? false);
   215→  const [name, setName] = useState('');
   216→  const [nameRaw, setNameRaw] = useState('');
   217→  const [village, setVillage] = useState('');
   218→  const [villageRaw, setVillageRaw] = useState('');
   219→  const [relationship, setRelationship] = useState('');
   220→  const [amountRaw, setAmountRaw] = useState('');
   221→  // नेवता/सामान mode — non-cash gift written in place of the amount
   222→  const [itemMode, setItemMode] = useState(false);
   223→  const [itemText, setItemText] = useState('');
   224→  const [date] = useState(todayISO()); // aaj ki tarikh — sirf history/record ke liye
   225→  const [nameErr, setNameErr] = useState('');
   226→  const [amountErr, setAmountErr] = useState('');
   227→  const [dup, setDup] = useState<string | null>(null);
   228→  const [busy, setBusy] = useState(false);
   229→
   230→  const rowRef = useRef<HTMLDivElement>(null);
   231→  const committing = useRef(false); // blur + Enter ek saath — double commit kabhi nahi
   232→  const autoTimer = useRef<ReturnType<typeof setTimeout> | null>(null);
   233→
   234→  const nameRef = useRef<HTMLInputElement>(null);
   235→  const villageMobileRef = useRef<HTMLInputElement>(null);
   236→  const villageDesktopRef = useRef<HTMLInputElement>(null);
   237→  const relationMobileRef = useRef<HTMLInputElement>(null);
   238→  const relationDesktopRef = useRef<HTMLInputElement>(null);
   239→  const amountRef = useRef<HTMLInputElement>(null);
   240→  const itemRef = useRef<HTMLInputElement>(null);
   241→
   242→  // focus the input that is actually visible (mobile vs desktop grids)
   243→  const focusFirstVisible = (...els: Array<HTMLInputElement | null>) => {
   244→    const el = els.find((x) => x && x.offsetParent !== null);
   245→    el?.focus();
   246→  };
   247→  const focusVillage = () => focusFirstVisible(villageMobileRef.current, villageDesktopRef.current);
   248→  const focusRelation = () => focusFirstVisible(relationMobileRef.current, relationDesktopRef.current);
   249→
   250→  // autoStart: khali line par click hua — naam wale khaane mein seedha pen rakho
   251→  useEffect(() => {
   252→    if (!autoStart) return;
   253→    const tm = setTimeout(() => nameRef.current?.focus(), 60);
   254→    return () => clearTimeout(tm);
   255→  }, []);
   256→
   257→  // ＋ button / page-advance asks this line to start writing.
   258→  // Only reacts to signal CHANGES (not the value at mount) so that flipping
   259→  // pages never pops the keyboard by itself.
   260→  const initialSignal = useRef(focusSignal);
   261→  useEffect(() => {
   262→    if (focusSignal === initialSignal.current) return;
   263→    initialSignal.current = focusSignal;
   264→    setActive(true);
   265→    const tm = setTimeout(() => nameRef.current?.focus(), 60);
   266→    return () => clearTimeout(tm);
   267→  }, [focusSignal]);
   268→
   269→  const amountNum = parseInt(amountRaw || '0', 10) || 0;
   270→
   271→  const findDuplicate = () =>
   272→    allEntries.find(
   273→      (e) =>
   274→        e.eventId === currentEventId &&
   275→        e.name.trim() !== '' &&
   276→        (normalizeName(e.name) === normalizeName(name) ||
   277→          (!!nameRaw && !!e.nameLatin && normalizeName(e.nameLatin) === normalizeName(nameRaw)))
   278→    );
   279→
   280→  const reset = () => {
   281→    setName('');
   282→    setNameRaw('');
   283→    setVillage('');
   284→    setVillageRaw('');
   285→    setRelationship('');
   286→    setAmountRaw('');
   287→    setItemMode(false);
   288→    setItemText('');
   289→    setNameErr('');
   290→    setAmountErr('');
   291→    setDup(null);
   292→    clear();
   293→  };
   294→
   295→  const doCommit = async (force = false, fromBlur = false, amountOverride?: number) => {
   296→    if (busy || committing.current) return;
   297→    const nm = name.trim();
   298→    const it = itemMode ? itemText.trim() : '';
   299→    const amt = amountOverride ?? amountNum; // chip-click ke saath state abhi update nahi hui hoti
   300→    let ok = true;
   301→    if (!nm) {
   302→      setNameErr(t('nameRequired'));
   303→      ok = false;
   304→    } else setNameErr('');
   305→    if (amt <= 0 && !it) {
   306→      setAmountErr(t('amountRequired'));
   307→      ok = false;
   308→    } else setAmountErr('');
   309→    if (!ok) {
   310→      // pen utha li (blur) → focus wapas nahi kheenchte, bas galti dikhti hai
   311→      if (!fromBlur) (nm ? (itemMode ? itemRef : amountRef) : nameRef).current?.focus();
   312→      return;
   313→    }
   314→    // duplicate — never blocks, just asks once (paper-note style)
   315→    if (!force && findDuplicate()) {
   316→      setDup(nm);
   317→      return;
   318→    }
   319→    committing.current = true;
   320→    if (autoTimer.current) {
   321→      clearTimeout(autoTimer.current);
   322→      autoTimer.current = null;
   323→    }
   324→    setDup(null);
   325→    setBusy(true);
   326→    try {
   327→      await addEntry({
   328→        name: nm,
   329→        village,
   330→        relationship,
   331→        amount: it ? 0 : amt,
   332→        item: it,
   333→        date,
   334→        note: '',
   335→        nameLatin: nameRaw !== nm ? nameRaw : '',
   336→        villageLatin: villageRaw !== village ? villageRaw : '',
   337→      });
   338→      // फटाफट — the next line is ready immediately
   339→      reset();
   340→      onCommitted();
   341→      // Enter se likha → agla naam turant (keyboard khula rahta hai);
   342→      // blur se likha → pen utha li thi, keyboard wapas nahi kheenchte
   343→      if (!fromBlur) setTimeout(() => nameRef.current?.focus(), 60);
   344→    } catch {
   345→      toast.error(t('saveError'));
   346→    } finally {
   347→      committing.current = false;
   348→      setBusy(false);
   349→    }
   350→  };
   351→
   352→  const cancelLine = () => {
   353→    if (autoTimer.current) clearTimeout(autoTimer.current);
   354→    reset();
   355→    setActive(false);
   356→    onCancel?.();
   357→  };
   358→
   359→  /** koi bhi field mein typing shuru → chip auto-commit ruk jao */
   360→  const armCancel = () => {
   361→    if (autoTimer.current) {
   362→      clearTimeout(autoTimer.current);
   363→      autoTimer.current = null;
   364→    }
   365→  };
   366→
   367→  /**
   368→   * Pen utha li (focus line se bahar) → line apne aap likh jaye.
   369→   * - sab khali → line chup chaap band
   370→   * - naam + रकम/सामान → seedha commit (जैसे Enter dabaya ho)
   371→   * - aadha-adhura → line khuli rahegi, galti dikh jayegi
   372→   */
   373→  const blurCommitCheck = () => {
   374→    if (committing.current) return;
   375→    const nm = name.trim();
   376→    const it = itemMode ? itemText.trim() : '';
   377→    const anyText = nm || village.trim() || relationship.trim() || amountRaw.trim() || it;
   378→    if (!anyText) {
   379→      setActive(false);
   380→      reset();
   381→      onCancel?.();
   382→      return;
   383→    }
   384→    if (nm && (amountNum > 0 || it)) {
   385→      void doCommit(false, true); // fromBlur — focus wapas nahi kheenchenge
   386→    } else {
   387→      // adhura — galti dikhao par keyboard wapas mat kheencho
   388→      if (!nm) setNameErr(t('nameRequired'));
   389→      if (amountNum <= 0 && !it) setAmountErr(t('amountRequired'));
   390→    }
   391→  };
   392→
   393→  const scheduleBlurCommit = () => {
   394→    setTimeout(() => {
   395→      if (focusLeftRow(rowRef.current)) blurCommitCheck();
   396→    }, 90);
   397→  };
   398→
   399→  // ── inactive: a quiet empty ruled line with a faint ✎ ──
   400→  if (!active) {
   401→    return (
   402→      <div
   403→        className={`ruled-row ${GRID} cursor-text items-center hover:bg-ink/[0.035]`}
   404→        onClick={() => {
   405→          setActive(true);
   406→          setTimeout(() => nameRef.current?.focus(), 40);
   407→        }}
   408→        role="button"
   409→        tabIndex={0}
   410→        aria-label={t('writeHere')}
   411→        onKeyDown={(e) => {
   412→          if (e.key === 'Enter' || e.key === ' ') {
   413→            e.preventDefault();
   414→            setActive(true);
   415→            setTimeout(() => nameRef.current?.focus(), 40);
   416→          }
   417→        }}
   418→      >
   419→        <div className="text-center text-[13px] text-ink-soft/70">{String(serial).padStart(2, '0')}</div>
   420→        <div className="min-w-0 select-none px-0.5 text-[15px] text-ink-soft/40">✎ {t('writeHere')}</div>
   421→        <div className="hidden sm:block" />
   422→        <div className="hidden sm:block" />
   423→        <div />
   424→        <div className="hidden sm:block" />
   425→      </div>
   426→    );
   427→  }
   428→
   429→  // ── active: write ON the line ──
   430→  return (
   431→    <div ref={rowRef} className={`ruled-row ${GRID} items-center bg-ink/[0.03]`}>
   432→      <div className="text-center text-[13px] text-ink-soft">{String(serial).padStart(2, '0')}</div>
   433→
   434→      {/* नाम (+ mobile mini-line for गाँव / रिश्ता) */}
   435→      <div className="min-w-0 px-0.5 py-1">
   436→        <div className="flex items-center gap-1">
   437→          <TranslitInput
   438→            ref={nameRef}
   439→            {...wire('name')}
   440→            className="cell-input text-[15px] font-semibold"
   441→            placeholder={t('namePh')}
   442→            value={name}
   443→            maxLength={60}
   444→            autoComplete="off"
   445→            enterKeyHint="next"
   446→            aria-label={t('name')}
   447→            script={script}
   448→            onValueChange={(v) => {
   449→              setName(v);
   450→              setNameErr('');
   451→            }}
   452→            onRawChange={setNameRaw}
   453→            onKeyDown={(e, fv) => {
   454→              armCancel();
   455→              if (e.key === 'Enter') {
   456→                e.preventDefault();
   457→                if (fv !== undefined) setName(fv);
   458→                focusVillage();
   459→              }
   460→              if (e.key === 'Escape') {
   461→                e.preventDefault();
   462→                cancelLine();
   463→              }
   464→            }}
   465→            onBlur={scheduleBlurCommit}
   466→          />
   467→          <FieldError msg={nameErr} />
   468→        </div>
   469→        {/* mobile: गाँव + रिश्ता on the second small line */}
   470→        <div className="mt-0.5 flex gap-2 sm:hidden">
   471→          <TranslitInput
   472→            ref={villageMobileRef}
   473→            {...wire('village')}
   474→            className="cell-input min-w-0 flex-1 text-[13px] text-ink-soft"
   475→            placeholder={t('villagePh')}
   476→            value={village}
   477→            maxLength={40}
   478→            autoComplete="off"
   479→            aria-label={t('village')}
   480→            script={script}
   481→            onValueChange={(v) => {
   482→              setVillage(v);
   483→              setVillageRaw(v);
   484→            }}
   485→            onKeyDown={(e) => {
   486→              armCancel();
   487→              if (e.key === 'Enter') {
   488→                e.preventDefault();
   489→                focusRelation();
   490→              }
   491→              if (e.key === 'Escape') cancelLine();
   492→            }}
   493→            onBlur={scheduleBlurCommit}
   494→          />
   495→          <TranslitInput
   496→            ref={relationMobileRef}
   497→            {...wire('relation')}
   498→            className="cell-input min-w-0 flex-1 text-[13px] text-ink-soft"
   499→            placeholder={t('relationPh')}
   500→            value={relationship}
   501→            maxLength={40}
   502→            autoComplete="off"
   503→            aria-label={t('relation')}
   504→            script={script}
   505→            onValueChange={setRelationship}
   506→            onKeyDown={(e) => {
   507→              armCancel();
   508→              if (e.key === 'Enter') {
   509→                e.preventDefault();
   510→                amountRef.current?.focus();
   511→              }
   512→              if (e.key === 'Escape') cancelLine();
   513→            }}
   514→            onBlur={scheduleBlurCommit}
   515→          />
   516→        </div>
   517→      </div>
   518→
   519→      {/* desktop: गाँव */}
   520→      <div className="hidden min-w-0 px-0.5 sm:block">
   521→        <TranslitInput
   522→          ref={villageDesktopRef}
   523→          {...wire('village')}
   524→          className="cell-input text-sm"
   525→          placeholder={t('villagePh')}
   526→          value={village}
   527→          maxLength={40}
   528→          autoComplete="off"
   529→          aria-label={t('village')}
   530→          script={script}
   531→          onValueChange={(v) => {
   532→            setVillage(v);
   533→            setVillageRaw(v);
   534→          }}
   535→          onKeyDown={(e) => {
   536→            armCancel();
   537→            if (e.key === 'Enter') {
   538→              e.preventDefault();
   539→              focusRelation();
   540→            }
   541→            if (e.key === 'Escape') cancelLine();
   542→          }}
   543→          onBlur={scheduleBlurCommit}
   544→        />
   545→      </div>
   546→
   547→      {/* desktop: रिश्ता */}
   548→      <div className="hidden min-w-0 px-0.5 sm:block">

   550→          ref={relationDesktopRef}
