    25→    25→function useSuggest() {
    26→    26→  const [sug, setSug] = useState<Suggestion[]>([]);
    27→    27→  const [sugField, setSugField] = useState<SugField>('name');
    28→    28→  const [pick, setPick] = useState<PickPayload | null>(null);
    29→    29→  const seq = useRef(0);
    30→    30→  const wire = (field: SugField) => ({
    31→    31→    onSuggest: (s: Suggestion[]) => {
    32→    32→      setSug(s);
    33→    33→      if (s.length) setSugField(field);
    34→    34→    },
    35→    35→    pick: sugField === field ? pick : null,
    36→    36→  });
    37→    37→  const choose = (sgn: Suggestion) => {
    38→    38→    seq.current += 1;
    39→    39→    setPick({ roman: sgn.roman, text: sgn.text, seq: seq.current });
    40→    40→    setSug([]);
    41→    41→  };
    42→    42→  return { sug, wire, choose, clear: () => setSug([]) };
    43→    43→}

   108→   108→export function WritingRow({
   109→   109→  serial,
   110→   110→  script,
   111→   111→  focusSignal,
   112→   112→  onCommitted,
   113→   113→}: {
   114→   114→  serial: number;
   115→   115→  script: InputScriptSetting;
   116→   116→  focusSignal: number;
   117→   117→  onCommitted: () => void;
   118→   118→}) {
   119→   119→  const t = useT();
   120→   120→  const allEntries = useDaali((s) => s.allEntries);
   121→   121→  const currentEventId = useDaali((s) => s.currentEventId);
   122→   122→  const addEntry = useDaali((s) => s.addEntry);
   123→   123→  const { sug, wire, choose, clear } = useSuggest();
   124→   124→
   125→   125→  const [active, setActive] = useState(false);
   126→   126→  const [name, setName] = useState('');
   127→   127→  const [nameRaw, setNameRaw] = useState('');
   128→   128→  const [village, setVillage] = useState('');
   129→   129→  const [villageRaw, setVillageRaw] = useState('');
   130→   130→  const [relationship, setRelationship] = useState('');
   131→   131→  const [amountRaw, setAmountRaw] = useState('');
   132→   132→  const [date, setDate] = useState(todayISO());
   133→   133→  const [nameErr, setNameErr] = useState('');
   134→   134→  const [amountErr, setAmountErr] = useState('');
   135→   135→  const [dup, setDup] = useState<string | null>(null);
   136→   136→  const [busy, setBusy] = useState(false);
   137→   137→
   138→   138→  const nameRef = useRef<HTMLInputElement>(null);
   139→   139→  const villageMobileRef = useRef<HTMLInputElement>(null);
   140→   140→  const villageDesktopRef = useRef<HTMLInputElement>(null);
   141→   141→  const relationMobileRef = useRef<HTMLInputElement>(null);
   142→   142→  const relationDesktopRef = useRef<HTMLInputElement>(null);
   143→   143→  const amountRef = useRef<HTMLInputElement>(null);
   144→   144→
   145→   145→  // focus the input that is actually visible (mobile vs desktop grids)
   146→   146→  const focusFirstVisible = (...els: Array<HTMLInputElement | null>) => {
   147→   147→    const el = els.find((x) => x && x.offsetParent !== null);
   148→   148→    el?.focus();
   149→   149→  };
   150→   150→  const focusVillage = () => focusFirstVisible(villageMobileRef.current, villageDesktopRef.current);
   151→   151→  const focusRelation = () => focusFirstVisible(relationMobileRef.current, relationDesktopRef.current);
   152→   152→
   153→   153→  // ＋ button / page-advance asks this line to start writing.
   154→   154→  // Only reacts to signal CHANGES (not the value at mount) so that flipping
   155→   155→  // pages never pops the keyboard by itself.
   156→   156→  const initialSignal = useRef(focusSignal);
   157→   157→  useEffect(() => {
   158→   158→    if (focusSignal === initialSignal.current) return;
   159→   159→    initialSignal.current = focusSignal;
   160→   160→    setActive(true);
   161→   161→    const tm = setTimeout(() => nameRef.current?.focus(), 60);
   162→   162→    return () => clearTimeout(tm);
   163→   163→  }, [focusSignal]);
   164→   164→
   165→   165→  const amountNum = parseInt(amountRaw || '0', 10) || 0;
   166→   166→
   167→   167→  const findDuplicate = () =>
   168→   168→    allEntries.find(
   169→   169→      (e) =>
   170→   170→        e.eventId === currentEventId &&
   171→   171→        e.name.trim() !== '' &&
   172→   172→        (normalizeName(e.name) === normalizeName(name) ||
   173→   173→          (!!nameRaw && !!e.nameLatin && normalizeName(e.nameLatin) === normalizeName(nameRaw)))
   174→   174→    );
   175→   175→
   176→   176→  const reset = () => {
   177→   177→    setName('');
   178→   178→    setNameRaw('');
   179→   179→    setVillage('');
   180→   180→    setVillageRaw('');
   181→   181→    setRelationship('');
   182→   182→    setAmountRaw('');
   183→   183→    setDate(todayISO());
   184→   184→    setNameErr('');
   185→   185→    setAmountErr('');
   186→   186→    setDup(null);
   187→   187→    clear();
   188→   188→  };
   189→   189→
   190→   190→  const doCommit = async (force = false) => {
   191→   191→    if (busy) return;
   192→   192→    const nm = name.trim();
   193→   193→    let ok = true;
   194→   194→    if (!nm) {
   195→   195→      setNameErr(t('nameRequired'));
   196→   196→      ok = false;
   197→   197→    } else setNameErr('');
   198→   198→    if (amountNum <= 0) {
   199→   199→      setAmountErr(t('amountRequired'));
   200→   200→      ok = false;
   201→   201→    } else setAmountErr('');
   202→   202→    if (!ok) {
   203→   203→      (nm ? amountRef : nameRef).current?.focus();
   204→   204→      return;
   205→   205→    }
   206→   206→    // duplicate — never blocks, just asks once (paper-note style)
   207→   207→    if (!force && findDuplicate()) {
   208→   208→      setDup(nm);
   209→   209→      return;
   210→   210→    }
   211→   211→    setDup(null);
   212→   212→    setBusy(true);
   213→   213→    try {
   214→   214→      await addEntry({
   215→   215→        name: nm,
   216→   216→        village,
   217→   217→        relationship,
   218→   218→        amount: amountNum,
   219→   219→        date,
   220→   220→        note: '',
   221→   221→        nameLatin: nameRaw !== nm ? nameRaw : '',
   222→   222→        villageLatin: villageRaw !== village ? villageRaw : '',
   223→   223→      });
   224→   224→      // फटाफट — the next line is ready immediately
   225→   225→      reset();
   226→   226→      onCommitted();
   227→   227→      setTimeout(() => nameRef.current?.focus(), 60);
   228→   228→    } catch {
   229→   229→      toast.error(t('saveError'));
   230→   230→    } finally {
   231→   231→      setBusy(false);
   232→   232→    }
   233→   233→  };
   234→   234→
   235→   235→  const cancelLine = () => {
   236→   236→    reset();
   237→   237→    setActive(false);
   238→   238→  };
   239→   239→
   240→   240→  // ── inactive: a quiet empty ruled line with a faint ✎ ──
   241→   241→  if (!active) {
   242→   242→    return (
   243→   243→      <div
   244→   244→        className={`ruled-row ${GRID} cursor-text items-center hover:bg-ink/[0.035]`}
   245→   245→        onClick={() => {
   246→   246→          setActive(true);
   247→   247→          setTimeout(() => nameRef.current?.focus(), 40);
   248→   248→        }}
   249→   249→        role="button"
   250→   250→        tabIndex={0}
   251→   251→        aria-label={t('writeHere')}
   252→   252→        onKeyDown={(e) => {
   253→   253→          if (e.key === 'Enter' || e.key === ' ') {
   254→   254→            e.preventDefault();
   255→   255→            setActive(true);
   256→   256→            setTimeout(() => nameRef.current?.focus(), 40);
   257→   257→          }
   258→   258→        }}
   259→   259→      >
   260→   260→        <div className="text-center text-[13px] text-ink-soft/70">{String(serial).padStart(2, '0')}</div>
   261→   261→        <div className="min-w-0 select-none px-0.5 text-[15px] text-ink-soft/40">✎ {t('writeHere')}</div>
   262→   262→        <div className="hidden sm:block" />
   263→   263→        <div className="hidden sm:block" />
   264→   264→        <div />
   265→   265→        <div className="hidden sm:block" />
   266→   266→      </div>
   267→   267→    );
   268→   268→  }
   269→   269→
   270→   270→  // ── active: write ON the line ──
   271→   271→  return (
   272→   272→    <div className={`ruled-row ${GRID} items-center bg-ink/[0.03]`}>
   273→   273→      <div className="text-center text-[13px] text-ink-soft">{String(serial).padStart(2, '0')}</div>
   274→   274→
   275→   275→      {/* नाम (+ mobile mini-line for गाँव / रिश्ता) */}
   276→   276→      <div className="min-w-0 px-0.5 py-1">
   277→   277→        <div className="flex items-center gap-1">
   278→   278→          <TranslitInput
   279→   279→            ref={nameRef}
   280→   280→            {...wire('name')}
   281→   281→            className="cell-input text-[15px] font-semibold"
   282→   282→            placeholder={t('namePh')}
   283→   283→            value={name}
   284→   284→            maxLength={60}
   285→   285→            autoComplete="off"
   286→   286→            enterKeyHint="next"
   287→   287→            aria-label={t('name')}
   288→   288→            script={script}
   289→   289→            onValueChange={(v) => {
   290→   290→              setName(v);
   291→   291→              setNameErr('');
   292→   292→            }}
   293→   293→            onRawChange={setNameRaw}
   294→   294→            onKeyDown={(e, fv) => {
   295→   295→              if (e.key === 'Enter') {
   296→   296→                e.preventDefault();
   297→   297→                if (fv !== undefined) setName(fv);
   298→   298→                focusVillage();
   299→   299→              }
   300→   300→              if (e.key === 'Escape') {
   301→   301→                e.preventDefault();
   302→   302→                cancelLine();
   303→   303→              }
   304→   304→            }}
   305→   305→          />
   306→   306→          <FieldError msg={nameErr} />
   307→   307→        </div>
   308→   308→        {/* mobile: गाँव + रिश्ता on the second small line */}
   309→   309→        <div className="mt-0.5 flex gap-2 sm:hidden">
   310→   310→          <TranslitInput
   311→   311→            ref={villageMobileRef}
   312→   312→            {...wire('village')}
   313→   313→            className="cell-input min-w-0 flex-1 text-[13px] text-ink-soft"
   314→   314→            placeholder={t('villagePh')}
   315→   315→            value={village}
   316→   316→            maxLength={40}
   317→   317→            autoComplete="off"
   318→   318→            aria-label={t('village')}
   319→   319→            script={script}
   320→   320→            onValueChange={(v) => {
   321→   321→              setVillage(v);
   322→   322→              setVillageRaw(v);
   323→   323→            }}
   324→   324→            onKeyDown={(e) => {
   325→   325→              if (e.key === 'Enter') {
   326→   326→                e.preventDefault();
   327→   327→                focusRelation();
   328→   328→              }
   329→   329→              if (e.key === 'Escape') cancelLine();
   330→   330→            }}
   331→   331→          />
   332→   332→          <TranslitInput
   333→   333→            ref={relationMobileRef}
   334→   334→            {...wire('relation')}
   335→   335→            className="cell-input min-w-0 flex-1 text-[13px] text-ink-soft"
   336→   336→            placeholder={t('relationPh')}
   337→   337→            value={relationship}
   338→   338→            maxLength={40}
   339→   339→            autoComplete="off"
   340→   340→            aria-label={t('relation')}
   341→   341→            script={script}
   342→   342→            onValueChange={setRelationship}
   343→   343→            onKeyDown={(e) => {
   344→   344→              if (e.key === 'Enter') {
   345→   345→                e.preventDefault();
   346→   346→                amountRef.current?.focus();
   347→   347→              }
   348→   348→              if (e.key === 'Escape') cancelLine();
   349→   349→            }}
   350→   350→          />
   351→   351→        </div>
   352→   352→      </div>
   353→   353→
   354→   354→      {/* desktop: गाँव */}
   355→   355→      <div className="hidden min-w-0 px-0.5 sm:block">
   356→   356→        <TranslitInput
   357→   357→          ref={villageDesktopRef}
   358→   358→          {...wire('village')}
   359→   359→          className="cell-input text-sm"
   360→   360→          placeholder={t('villagePh')}
   361→   361→          value={village}
   362→   362→          maxLength={40}
   363→   363→          autoComplete="off"
   364→   364→          aria-label={t('village')}
   365→   365→          script={script}
   366→   366→          onValueChange={(v) => {
   367→   367→            setVillage(v);
   368→   368→            setVillageRaw(v);
   369→   369→          }}
   370→   370→          onKeyDown={(e) => {
   371→   371→            if (e.key === 'Enter') {
   372→   372→              e.preventDefault();
   373→   373→              focusRelation();

   375→   375→            if (e.key === 'Escape') cancelLine();
   376→   376→          }}
   377→   377→        />
   378→   378→      </div>

   380→   380→      {/* desktop: रिश्ता */}
   381→   381→      <div className="hidden min-w-0 px-0.5 sm:block">
   382→   382→        <TranslitInput
   383→   383→          ref={relationDesktopRef}
   384→   384→          {...wire('relation')}
   385→   385→          className="cell-input text-sm"
   386→   386→          placeholder={t('relationPh')}
   387→   387→          value={relationship}
   388→   388→          maxLength={40}
   389→   389→          autoComplete="off"
   390→   390→          aria-label={t('relation')}
   391→   391→          script={script}
   392→   392→          onValueChange={setRelationship}
   393→   393→          onKeyDown={(e) => {
   394→   394→            if (e.key === 'Enter') {
   395→   395→              e.preventDefault();
   396→   396→              amountRef.current?.focus();

   400→   400→        />
   401→   401→      </div>
   402→   402→
   403→   403→      {/* रकम */}
   404→   404→      <div className="px-0.5">
   405→   405→        <div className="flex items-center justify-end gap-0.5">
   406→   406→          <span className="font-hand text-base font-bold text-ink-soft">₹</span>
   407→   407→          <input
   408→   408→            ref={amountRef}
   409→   409→            className="cell-input w-full text-right text-[15px] font-bold tabular-nums"
   410→   410→            style={{ fontVariantNumeric: 'tabular-nums' }}
   411→   411→            inputMode="numeric"
   412→   412→            pattern="[0-9]*"
   413→   413→            autoComplete="off"
   414→   414→            placeholder="0"
   415→   415→            value={amountRaw}

   426→   426→                doCommit();

   430→   430→                cancelLine();

   432→   432→            }}
   433→   433→          />
   434→   434→          <FieldError msg={amountErr} />
   435→   435→        </div>
   436→   436→      </div>
   437→   437→
   438→   438→      {/* desktop: तारीख */}
   439→   439→      <div className="hidden px-0.5 sm:block">
   440→   440→        <input

   458→   458→      {dup ? (

   488→   488→        />
   489→   489→      )}
   490→   490→    </div>
   491→   491→  );
